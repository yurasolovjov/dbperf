WITH locks AS (
  SELECT pid,
         locktype,
         mode,
         relation,
         page,
         tuple,
         virtualxid,
         transactionid,
         classid,
         objid,
         objsubid,
         granted
         FROM pg_locks
         WHERE locktype IN (SELECT locktype
                            FROM pg_locks
                            WHERE granted = 'f')
)
SELECT extract(epoch FROM clock_timestamp())::int unix_time,
       a.pid,
       host(a.client_addr) client_addr,
       a.usename,
       a.datid::int,
       a.datname,
       a.state,
       a.query,
       a.wait_event_type,
       a.wait_event,
       extract(EPOCH FROM a.xact_start)::int xact_start,
       extract(EPOCH FROM a.state_change)::int state_change,
       extract(EPOCH FROM a.query_start)::int query_start,
       l.locktype,
       l.mode,
       l.relation,
       l.page,
       l.tuple,
       l.virtualxid,
       l.transactionid::text,
       l.classid,
       l.objid,
       l.objsubid,
       l.granted
FROM pg_stat_activity a
JOIN pg_database d ON (a.datid = d.oid)
LEFT JOIN locks l USING (pid)
WHERE a.pid <> $1
  AND (a.state IN ('active', 'fastpath function call') OR a.pid IN (SELECT pid FROM locks));
