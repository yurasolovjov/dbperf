       host(client_addr) client_addr,
       usename,
       datid::int,
       datname,
       state,
       query,
       wait_event_type,
       wait_event,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL,
       NULL
FROM pg_stat_activity
WHERE state IN ('active', 'fastpath function call')
